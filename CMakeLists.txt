add_library(send_airplay2 src/http_range.cpp src/media_server.cpp src/discovery.cpp src/discovery_core.cpp src/discovery_transport.cpp ${sap2_pairing_sources} ${sap2_verification_sources} ${sap2_setup_sources} ${sap2_receiver_sources})
target_link_libraries(send_airplay2 PRIVATE OpenSSL::Crypto ${sap2_botan_target} Threads::Threads ${CMAKE_DL_LIBS} Boost::beast Boost::asio)
target_compile_features(send_airplay2 PUBLIC cxx_std_17)
target_include_directories(send_airplay2 PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
set_target_properties(send_airplay2 PROPERTIES CXX_EXTENSIONS OFF)
if(BUILD_SHARED_LIBS)
  target_compile_definitions(send_airplay2 PUBLIC SAP2_SHARED PRIVATE SAP2_BUILDING)
endif()
if(WIN32)
  target_compile_definitions(send_airplay2 PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(send_airplay2 PRIVATE ws2_32 iphlpapi)
endif()
function(sap2_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
sap2_warnings(send_airplay2)
# Host auth is private C++; compile it directly, without exporting secrets through the DLL ABI.
add_executable(airplay2-cli src/cli.cpp src/auth_cli.cpp src/serve_cli.cpp src/file_media_source.cpp ${sap2_auth_sources})
target_include_directories(airplay2-cli PRIVATE src)
target_link_libraries(airplay2-cli PRIVATE send_airplay2 OpenSSL::Crypto ${sap2_botan_target} Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(airplay2-cli PRIVATE ws2_32 advapi32)
endif()
sap2_warnings(airplay2-cli)
