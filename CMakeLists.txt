cmake_minimum_required(VERSION 3.20)
project(send_airplay2 VERSION 0.1.0 LANGUAGES C CXX)

add_library(send_airplay2 src/http_range.cpp)
target_compile_features(send_airplay2 PRIVATE cxx_std_17)
target_include_directories(send_airplay2 PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
set_target_properties(send_airplay2 PROPERTIES CXX_EXTENSIONS OFF)
if(BUILD_SHARED_LIBS)
  target_compile_definitions(send_airplay2 PUBLIC SAP2_SHARED PRIVATE SAP2_BUILDING)
endif()
if(MSVC)
  target_compile_options(send_airplay2 PRIVATE /W4 /WX)
else()
  target_compile_options(send_airplay2 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(range_tests tests/range_tests.cpp)
  target_link_libraries(range_tests PRIVATE send_airplay2)
  target_compile_features(range_tests PRIVATE cxx_std_17)
  add_test(NAME range_tests COMMAND range_tests)
  add_executable(c_abi_smoke tests/c_abi_smoke.c)
  target_link_libraries(c_abi_smoke PRIVATE send_airplay2)
  set_target_properties(c_abi_smoke PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME c_abi_smoke COMMAND c_abi_smoke)
endif()
