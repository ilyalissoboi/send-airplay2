cmake_minimum_required(VERSION 3.20)
project(send_airplay2 VERSION 0.1.0 LANGUAGES C CXX)

find_package(OpenSSL 3.5 REQUIRED COMPONENTS Crypto)
find_package(Botan 3.12 CONFIG REQUIRED COMPONENTS ffi srp6 sha2_64 system_rng)
if(TARGET botan::botan)
  set(sap2_botan_target botan::botan)
else()
  set(sap2_botan_target botan::botan-static)
endif()
find_package(Threads REQUIRED)
set(sap2_pairing_sources src/pairing_tlv.cpp src/control_crypto.cpp src/control_records.cpp)
set(sap2_verification_sources src/identity_crypto.cpp src/pair_verify.cpp)
set(sap2_setup_sources src/pair_setup_crypto.cpp src/pair_setup.cpp)
add_library(send_airplay2 src/http_range.cpp src/discovery.cpp src/discovery_core.cpp src/discovery_transport.cpp ${sap2_pairing_sources} ${sap2_verification_sources} ${sap2_setup_sources})
target_link_libraries(send_airplay2 PRIVATE OpenSSL::Crypto ${sap2_botan_target} Threads::Threads ${CMAKE_DL_LIBS})
target_compile_features(send_airplay2 PUBLIC cxx_std_17)
target_include_directories(send_airplay2 PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
set_target_properties(send_airplay2 PROPERTIES CXX_EXTENSIONS OFF)
if(BUILD_SHARED_LIBS)
  target_compile_definitions(send_airplay2 PUBLIC SAP2_SHARED PRIVATE SAP2_BUILDING)
endif()
if(WIN32)
  target_link_libraries(send_airplay2 PRIVATE ws2_32 iphlpapi)
endif()
function(sap2_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
sap2_warnings(send_airplay2)
add_executable(airplay2-cli src/cli.cpp)
target_link_libraries(airplay2-cli PRIVATE send_airplay2)
sap2_warnings(airplay2-cli)
